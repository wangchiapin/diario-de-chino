        && request.auth.token.email_verified == true
        && get(/databases/$(database)/documents/portal/$(tid)/roster/$(sid)).data.email == request.auth.token.email
        && resource.data.status == 'todo'
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['answers', 'status', 'submittedAt'])
        && request.resource.data.status in ['todo', 'submitted'];
    }

// lists：老師在後台為每位學生補充、修改、隱藏的單字和語法點（練習單的單字會自動匯入，不存在這裡）。

    // 單字／語法清單：老師寫；老師和「信箱已驗證、而且就是這位學生」的人可讀

// members：學生名單鏡像（文件代號＝學生電子郵件）。老師在後台確認學生信箱時自動寫入；規則用它判斷「這個人是不是老師的學生」。
// board：公布欄，老師寫、學生讀。 wall：公開留言，學生寫（只能用自己的名字和代號）、所有學生讀，自己的可以自己刪。
// dm：私訊，只有老師和那位學生讀；學生只能寫 from=='s'、老師寫 from=='t'；寫了不能改。

    // 學生名單鏡像：只有老師能讀寫（學生不需要讀它，規則內部用 exists／get 查）

// 第十批（能力分級測驗結果、老師備註）：只要「新增」下面這兩段，貼在其他 match 旁邊（並排）。不用動前面的任何一段。
// placements：每次測驗存一筆。兩種來源：
//   src=='portal'：學生頁測完自動存，要登入、信箱已驗證、是老師的學生，而且 sid 要是自己的（和私訊一樣的判斷）。
//   src=='web'：獨立網頁（沒登入）測完自動存，只能「新增」、不能讀改刪；欄位、字數、數字範圍都有限制，避免亂填。
// 讀：老師全部；學生只能讀 sid 是自己的那幾筆。刪：只有老師。寫了不能改。
// notes：老師專用備註（每位學生一份，文件代號＝學生代號 sid），學生讀不到。
// ===================================================================

    match /portal/{tid}/placements/{id} {
      allow read: if request.auth != null && (
        request.auth.uid == tid ||
        (request.auth.token.email_verified == true &&
         exists(/databases/$(database)/documents/portal/$(tid)/members/$(request.auth.token.email)) &&
         resource.data.sid == get(/databases/$(database)/documents/portal/$(tid)/members/$(request.auth.token.email)).data.sid));
      allow create: if
        request.resource.data.keys().hasOnly(['src', 'sid', 'name', 'mode', 'ts', 'reached', 'cefr', 'tocfl', 'right', 'total', 'by', 'py']) &&
        request.resource.data.name is string && request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 40 &&
        request.resource.data.mode in ['quick', 'full'] &&
        request.resource.data.reached is int && request.resource.data.reached >= 0 && request.resource.data.reached <= 6 &&
        request.resource.data.cefr is string && request.resource.data.cefr.size() <= 6 &&
        request.resource.data.tocfl is string && request.resource.data.tocfl.size() <= 5 &&
        request.resource.data.right is int && request.resource.data.right >= 0 && request.resource.data.right <= 60 &&
        request.resource.data.total is int && request.resource.data.total >= 1 && request.resource.data.total <= 60 &&
        request.resource.data.right <= request.resource.data.total &&
        request.resource.data.by is map && request.resource.data.by.size() <= 6 &&
        request.resource.data.py is bool &&
        request.resource.data.ts is number &&
        request.resource.data.ts > request.time.toMillis() - 86400000 && request.resource.data.ts < request.time.toMillis() + 86400000 &&
        ((request.resource.data.src == 'web' && !('sid' in request.resource.data)) ||
         (request.resource.data.src == 'portal' && request.auth != null &&
          request.auth.token.email_verified == true &&
          exists(/databases/$(database)/documents/portal/$(tid)/members/$(request.auth.token.email)) &&
          request.resource.data.sid == get(/databases/$(database)/documents/portal/$(tid)/members/$(request.auth.token.email)).data.sid));
      allow update: if false;
      allow delete: if request.auth != null && request.auth.uid == tid;
    }

      allow read, write: if request.auth != null && request.auth.uid == tid;
